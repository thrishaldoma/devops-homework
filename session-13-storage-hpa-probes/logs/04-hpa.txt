# ===== 2. configure the HPA (min 1, max 5, target 50% CPU) =====
$ cat session-13-storage-hpa-probes/04-hpa/hpa.yaml
apiVersion: autoscaling/v2
kind: HorizontalPodAutoscaler
metadata:
  name: hpa-demo
spec:
  scaleTargetRef:
    apiVersion: apps/v1
    kind: Deployment
    name: hpa-demo
  minReplicas: 1
  maxReplicas: 5
  metrics:
    - type: Resource
      resource:
        name: cpu
        target:
          type: Utilization
          averageUtilization: 50

$ kubectl apply -f session-13-storage-hpa-probes/04-hpa/hpa.yaml
horizontalpodautoscaler.autoscaling/hpa-demo created

# until ~375s - a gap of about 5 minutes. That is the default scale-DOWN

# one 15s sync. Kubernetes deliberately scales out fast and scales in slowly, so a
