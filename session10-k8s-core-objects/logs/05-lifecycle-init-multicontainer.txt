# ===== 9. Graceful termination: SIGTERM trap + terminationGracePeriodSeconds =====
$ kubectl apply -f session10-k8s-core-objects/pod-lifecycle/12-termination.yaml
pod/lifecycle-termination created

$ kubectl wait --for=condition=Ready pod/lifecycle-termination --timeout=120s
pod/lifecycle-termination condition met

# timing the delete - the trap sleeps 10s before exiting, so deletion is NOT instant
$ time kubectl delete -f session10-k8s-core-objects/pod-lifecycle/12-termination.yaml
pod "lifecycle-termination" deleted from default namespace
  0.02s user 0.02s system 0% cpu 11.453 total
