# ########## DRILL 2: immutable selector / template label mismatch ##########
$ grep -E 'matchLabels|app:' session10-k8s-core-objects/troubleshooting/selector-mismatch.yaml
# spec.selector.matchLabels (app: backend) does not match
# spec.template.metadata.labels (app: frontend).
    matchLabels:
      app: backend      # <-- selector says "backend"
        app: frontend   # <-- template says "frontend"  => REJECTED

# applying it - the API server rejects the object at admission time:
$ kubectl apply -f session10-k8s-core-objects/troubleshooting/selector-mismatch.yaml
The Deployment "selector-error-demo" is invalid: spec.template.metadata.labels: Invalid value: {"app":"frontend"}: `selector` does not match template `labels`

# FIX: make spec.template.metadata.labels match spec.selector.matchLabels
$ diff session10-k8s-core-objects/troubleshooting/selector-mismatch.yaml session10-k8s-core-objects/troubleshooting/selector-fixed.yaml

---
> # Drill 2 - CORRECTED version (selector == template labels).
19c19
<         app: frontend   # <-- template says "frontend"  => REJECTED
---
>         app: backend    # <-- FIXED: now matches the selector
