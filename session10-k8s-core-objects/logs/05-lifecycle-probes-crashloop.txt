# ===== 1. Pending: scheduler cannot satisfy a 512Gi memory request =====
$ kubectl apply -f session10-k8s-core-objects/pod-lifecycle/02-pending.yaml
pod/lifecycle-pending created

$ kubectl get pod lifecycle-pending
NAME                READY   STATUS    RESTARTS   AGE
lifecycle-pending   0/1     Pending   0          8s

$ kubectl describe pod lifecycle-pending | grep -A 5 'Events:'
Events:
  Type     Reason            Age   From               Message
  ----     ------            ----  ----               -------
  Warning  FailedScheduling  8s    default-scheduler  0/2 nodes are available: 2 Insufficient cpu, 2 Insufficient memory. preemption: 0/2 nodes are available: 2 Preemption is not helpful for scheduling.

$ kubectl delete -f session10-k8s-core-objects/pod-lifecycle/02-pending.yaml --wait=false
pod "lifecycle-pending" deleted from default namespace

# ===== 2. CrashLoopBackOff: container exits 1 immediately; kubelet backs off exponentially =====
$ kubectl apply -f session10-k8s-core-objects/pod-lifecycle/05-crashloopbackoff.yaml
pod/lifecycle-crashloop created

$ while true; do kubectl get pod lifecycle-crashloop --no-headers; sleep 1; done
lifecycle-crashloop  0/1  ContainerCreating  restarts=0
lifecycle-crashloop  0/1  Error  restarts=0
lifecycle-crashloop  0/1  Error  restarts=1
lifecycle-crashloop  0/1  Error  restarts=2
lifecycle-crashloop  0/1  Error  restarts=3
lifecycle-crashloop  0/1  CrashLoopBackOff  restarts=3

$ kubectl logs lifecycle-crashloop
starting up...
FATAL: cannot reach database

$ kubectl describe pod lifecycle-crashloop | grep -A 8 'Events:' | tail -4
  Warning  BackOff    45s (x3 over 80s)  kubelet            spec.containers{crasher}: Back-off restarting failed container crasher in pod lifecycle-crashloop_default(9a79f4f4-6015-426e-b5d3-ff868b0f401d)
  Normal   Pulled     0s (x5 over 82s)   kubelet            spec.containers{crasher}: Container image "busybox:1.36" already present on machine and can be accessed by the pod
  Normal   Created    0s (x5 over 82s)   kubelet            spec.containers{crasher}: Container created
  Normal   Started    0s (x5 over 82s)   kubelet            spec.containers{crasher}: Container started

$ kubectl delete -f session10-k8s-core-objects/pod-lifecycle/05-crashloopbackoff.yaml --wait=false
pod "lifecycle-crashloop" deleted from default namespace

# ===== 3. Readiness probe: Running != Ready =====

# ===== 4. Liveness probe: automated self-healing restart =====
$ kubectl apply -f session10-k8s-core-objects/pod-lifecycle/08-liveness.yaml
pod/lifecycle-liveness created

$ while true; do kubectl get pod lifecycle-liveness --no-headers; sleep 2; done
lifecycle-liveness  0/1  ContainerCreating  RESTARTS=0
lifecycle-liveness  1/1  Running  RESTARTS=0
lifecycle-liveness  1/1  Running  RESTARTS=1

$ kubectl describe pod lifecycle-liveness | grep -E 'Liveness|Killing|Unhealthy' | head -5
    Liveness:       exec [cat /tmp/healthy] delay=5s timeout=1s period=5s successThreshold=1 failureThreshold=3
  Warning  Unhealthy  31s (x3 over 41s)  kubelet            spec.containers{app}: Liveness probe failed: cat: can't open '/tmp/healthy': No such file or directory
  Normal   Killing    31s                kubelet            spec.containers{app}: Container app failed liveness probe, will be restarted

$ kubectl delete -f session10-k8s-core-objects/pod-lifecycle/08-liveness.yaml --wait=false
pod "lifecycle-liveness" deleted from default namespace

# ===== 5. Startup probe: protects a slow-booting app from the liveness probe =====
$ kubectl apply -f session10-k8s-core-objects/pod-lifecycle/09-startup.yaml
pod/lifecycle-startup created

$ while true; do kubectl get pod lifecycle-startup --no-headers; sleep 3; done
lifecycle-startup  READY=0/1  ContainerCreating  RESTARTS=0
lifecycle-startup  READY=0/1  Running  RESTARTS=0
lifecycle-startup  READY=1/1  Running  RESTARTS=0

# RESTARTS stayed 0 through a 30s boot: the startup probe suspends liveness until it first succeeds
$ kubectl describe pod lifecycle-startup | grep -E 'Startup:|Liveness:'
    Liveness:       exec [cat /tmp/started] delay=0s timeout=1s period=5s successThreshold=1 failureThreshold=3
    Startup:        exec [cat /tmp/started] delay=0s timeout=1s period=5s successThreshold=1 failureThreshold=30

$ kubectl delete -f session10-k8s-core-objects/pod-lifecycle/09-startup.yaml --wait=false
pod "lifecycle-startup" deleted from default namespace
