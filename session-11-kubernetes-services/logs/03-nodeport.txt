# the nodePort is open on EVERY node, including one that is not running a pod
# of this app - kube-proxy forwards across nodes. Proof from inside the cluster:
# (allow a few seconds first: kube-proxy must program iptables on each node)
$ kubectl exec curl-client -- curl -s -o /dev/null -w 'node 192.168.49.2:30080 -> HTTP %{http_code}\n' --max-time 5 http://192.168.49.2:30080
node 192.168.49.2:30080 -> HTTP 200

$ kubectl exec curl-client -- curl -s -o /dev/null -w 'node 192.168.49.3:30080 -> HTTP %{http_code}\n' --max-time 5 http://192.168.49.3:30080
node 192.168.49.3:30080 -> HTTP 200

# ===== reaching the NodePort from the macOS HOST =====
# Direct access to the node IP does not work with the Docker driver (see Task 12):
$ curl --connect-timeout 3 -s -o /dev/null -w 'host -> 192.168.49.2:30080 = HTTP %{http_code}\n' http://192.168.49.2:30080 || echo 'host -> 192.168.49.2:30080 = FAILED (unreachable)'
host -> 192.168.49.2:30080 = HTTP 000
host -> 192.168.49.2:30080 = FAILED (unreachable)
