# Traffic to the internal name reaches the external host. -k is required because
# the TLS certificate is issued for api.github.com, not for our alias:
$ kubectl exec dns-test-client -- curl -s -o /dev/null -w 'https://external-database-service -> HTTP %{http_code}\n' --max-time 20 -k https://external-database-service
https://external-database-service -> HTTP 400

# without -k the handshake fails with exit 60 (certificate name mismatch):
$ kubectl exec dns-test-client -- curl -s --max-time 20 https://external-database-service 2>&1 | head -2; echo "curl exit code: $?"
command terminated with exit code 60
curl exit code: 0

# HTTP 400 proves we really reached GitHub (it rejects the unexpected Host header).
# Sending the correct Host header through the same alias returns a real API response:
$ kubectl exec curl-client -- curl -sS --max-time 25 -k -H 'Host: api.github.com' -o /tmp/gh.json -w 'HTTP %{http_code}, %{size_download} bytes from %{remote_ip}\n' https://external-database-service
