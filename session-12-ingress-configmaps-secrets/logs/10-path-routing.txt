$ kubectl describe ingress yatri-ingress

# ===== '/' goes to the FRONTEND (nginx) =====
$ curl -s -H 'Host: yatri.local' http://127.0.0.1:18080/ | grep -i '<title>'
<title>Welcome to nginx!</title>

# ===== '/api/' goes to the BACKEND (python) =====
$ curl -s -H 'Host: yatri.local' http://127.0.0.1:18080/api/
Yatri Backend API
path: /
--------------------------------
ENVIRONMENT: production
LOG_LEVEL: INFO
PORT: 8080
DEFAULT_CURRENCY: INR
MAX_BOOKING_DAYS: 90
POSTGRES_USER: yatri_admin
POSTGRES_PASSWORD: secretpassword

# ===== the rewrite-target strips the /api prefix before forwarding =====
# the backend reports the path it actually received:
$ curl -s -H 'Host: yatri.local' http://127.0.0.1:18080/api/bookings | grep '^path:'
path: /bookings

# client asked for /api/bookings; the backend saw /bookings (rewrite-target: /$2)
# ===== a host with no matching rule is not routed =====
$ curl -s -o /dev/null -w 'Host: unknown.local -> HTTP %{http_code}\n' -H 'Host: unknown.local' http://127.0.0.1:18080/
Host: unknown.local -> HTTP 404

$ curl -s -o /dev/null -w 'Host: yatri.local   -> HTTP %{http_code}\n' -H 'Host: yatri.local' http://127.0.0.1:18080/
Host: yatri.local   -> HTTP 200
