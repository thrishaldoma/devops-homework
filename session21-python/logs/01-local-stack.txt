# --- PUT and DELETE exercised against a throwaway task ---
$ curl -s -X POST http://localhost:8000/api/tasks -H 'Content-Type: application/json' -d '{"title":"Temporary task","priority":"LOW","assignee":"demo","status":"TODO"}' | python3 -m json.tool
{
    "title": "Temporary task",
    "description": "",
    "priority": "LOW",
    "status": "TODO",
    "assignee": "demo",
    "id": 7,

$ curl -s -X PUT http://localhost:8000/api/tasks/7 -H 'Content-Type: application/json' -d '{"status":"DONE"}' | python3 -c "import json,sys; t=json.load(sys.stdin); print('id=%s status is now %s' % (t['id'], t['status']))"
