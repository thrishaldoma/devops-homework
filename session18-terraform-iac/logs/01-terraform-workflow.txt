# ===== state: the mapping between config and real infrastructure =====
$ terraform -chdir=/Users/thrishaldoma/devops-homework/session18-terraform-iac/terraform-s3-demo state list
aws_s3_bucket.demo
aws_s3_bucket_public_access_block.demo
aws_s3_bucket_server_side_encryption_configuration.demo
aws_s3_bucket_versioning.demo

# The state file records resource IDs. Deleting it would orphan the real
# resources: Terraform would try to CREATE them again and fail. This is why
# production state lives in a remote backend (S3 + DynamoDB lock), never on a
# laptop and never in Git.
# 
# ===== idempotency: a second plan finds nothing to do =====
$ terraform -chdir=/Users/thrishaldoma/devops-homework/session18-terraform-iac/terraform-s3-demo plan -no-color 2>&1 | grep -E 'No changes|Plan:'
No changes. Your infrastructure matches the configuration.

# That is the declarative property: the config describes a DESIRED state, so
# re-running apply is a no-op instead of creating a second bucket.
# 
# ===== drift detection: change reality behind Terraform's back =====
# delete the bucket directly through the S3 API, bypassing Terraform:
$ curl -s -X DELETE -o /dev/null -w 'DELETE bucket -> HTTP %{http_code}\n' http://localhost:4566/yatri-session18-demo
DELETE bucket -> HTTP 204

$ curl -s -o /dev/null -w 'bucket now        -> HTTP %{http_code} (404 = gone)\n' http://localhost:4566/yatri-session18-demo
bucket now        -> HTTP 404 (404 = gone)

# Terraform's state still believes it exists. Ask it to re-plan:
$ terraform -chdir=/Users/thrishaldoma/devops-homework/session18-terraform-iac/terraform-s3-demo plan -no-color 2>&1 | grep -E 'has been deleted|will be created|Plan:' | head -5
  # aws_s3_bucket.demo has been deleted
  # aws_s3_bucket.demo will be created
  # aws_s3_bucket_public_access_block.demo will be created
  # aws_s3_bucket_server_side_encryption_configuration.demo will be created
  # aws_s3_bucket_versioning.demo will be created

# It refreshed state, saw the resource was gone, and planned to recreate it.
# That refresh step is why you never click in the console on a Terraform-managed
# resource: Terraform will quietly put it back on the next apply.
$ terraform -chdir=/Users/thrishaldoma/devops-homework/session18-terraform-iac/terraform-s3-demo apply -auto-approve -no-color 2>&1 | tail -3
bucket_name = "yatri-session18-demo"
bucket_region = "ap-south-1"
versioning_status = "Enabled"

$ curl -s -o /dev/null -w 'bucket restored   -> HTTP %{http_code}\n' http://localhost:4566/yatri-session18-demo
bucket restored   -> HTTP 200
